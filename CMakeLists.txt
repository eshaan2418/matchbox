cmake_minimum_required(VERSION 3.25)
project(matchbox LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(MATCHBOX_NATIVE "Build with -mcpu/-march=native" OFF)
set(MATCHBOX_SANITIZE "" CACHE STRING "Sanitizers: address,undefined | thread | empty")
option(MATCHBOX_FUZZ "Build libFuzzer harnesses" OFF)
option(MATCHBOX_BENCH "Build benchmarks" ON)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror -Wconversion -Wno-sign-conversion)
if(MATCHBOX_NATIVE)
  if(APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    add_compile_options(-mcpu=native)
  else()
    add_compile_options(-march=native)
  endif()
endif()
if(MATCHBOX_SANITIZE MATCHES "thread")
  add_compile_definitions(MATCHBOX_NO_ALLOC_HOOK)
endif()
if(MATCHBOX_SANITIZE)
  add_compile_options(-fsanitize=${MATCHBOX_SANITIZE} -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${MATCHBOX_SANITIZE})
endif()
find_package(Threads REQUIRED)

add_library(matchbox src/journal.cpp src/net.cpp src/gen.cpp)
target_include_directories(matchbox PUBLIC include)
target_link_libraries(matchbox PUBLIC Threads::Threads)

add_executable(matchbox-cli tools/matchbox.cpp src/alloc_counter.cpp)
set_target_properties(matchbox-cli PROPERTIES OUTPUT_NAME matchbox)
target_compile_options(matchbox-cli PRIVATE -fno-exceptions -fno-rtti)
target_link_libraries(matchbox-cli PRIVATE matchbox)

add_executable(mbload tools/mbload.cpp)
target_compile_options(mbload PRIVATE -fno-exceptions -fno-rtti)
target_link_libraries(mbload PRIVATE matchbox)

include(FetchContent)
set(FETCHCONTENT_QUIET ON)
FetchContent_Declare(googletest
  URL https://github.com/google/googletest/archive/refs/tags/v1.17.0.tar.gz
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE SYSTEM)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
FetchContent_Declare(rapidcheck
  GIT_REPOSITORY https://github.com/emil-e/rapidcheck.git
  GIT_TAG 2c3c4365aca21ef4e612768fdc95b1ce8b39a651 SYSTEM)
set(RC_ENABLE_GTEST ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest rapidcheck)
foreach(t gtest gtest_main gmock rapidcheck)
  if(TARGET ${t})
    get_target_property(_type ${t} TYPE)
    if(NOT _type STREQUAL "INTERFACE_LIBRARY")
      target_compile_options(${t} PRIVATE -w)
    endif()
  endif()
endforeach()

enable_testing()
add_executable(matchbox-tests tests/test_rules.cpp tests/test_differential.cpp tests/test_property.cpp
  tests/test_journal.cpp tests/test_protocol.cpp tests/test_spsc.cpp tests/test_gateway.cpp)
target_link_libraries(matchbox-tests PRIVATE matchbox GTest::gtest_main rapidcheck rapidcheck_gtest)
include(GoogleTest)
gtest_discover_tests(matchbox-tests DISCOVERY_TIMEOUT 60)

add_executable(matchbox-alloc-test tests/test_alloc.cpp src/alloc_counter.cpp)
target_link_libraries(matchbox-alloc-test PRIVATE matchbox GTest::gtest_main)
gtest_discover_tests(matchbox-alloc-test DISCOVERY_TIMEOUT 60)

if(MATCHBOX_BENCH AND NOT MATCHBOX_SANITIZE)
  add_executable(matchbox-bench bench/engine_bench.cpp src/alloc_counter.cpp)
  target_compile_options(matchbox-bench PRIVATE -fno-exceptions -fno-rtti)
  target_link_libraries(matchbox-bench PRIVATE matchbox)
endif()

if(MATCHBOX_FUZZ)
  foreach(f fuzz_engine fuzz_protocol)
    add_executable(${f} fuzz/${f}.cpp)
    target_compile_options(${f} PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_options(${f} PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_libraries(${f} PRIVATE matchbox)
  endforeach()
endif()
